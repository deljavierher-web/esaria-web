ocr.py // EXTRACTOR REGEX ESL10-L31

          
def extract_plate_from_text(text: str) -> Optional[str]:
    text = text.upper().strip()
    # 1) Formato moderno: 4 dígitos + 3 consonantes
    m_mod = re.search(
        r'([0-9]{4})\s*[- ]?\s*([BCDFGHJKLMNPRSTVWXYZ]{3})',
        text
    )
    if m_mod:
        return f"{m_mod.group(1)}{m_mod.group(2)}"
    # 2) Formato provincial (ej. VA 1234 AB)
    m_prv = re.search(
        r'([A-Z]{1,2})\s*[- ]?\s*([0-9]{4})\s*[- ]?\s*([A-Z]{1,2})',
        text
    )
    if m_prv:
        return f"{m_prv.group(1)}{m_prv.group(2)}{m_prv.group(3)}"
